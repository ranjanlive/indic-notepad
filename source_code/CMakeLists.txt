cmake_minimum_required(VERSION 3.24)
project(IndicNotePad VERSION 0.1.0 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Indic NotePad currently requires Windows x64 and MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /permissive- /EHsc)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
include(FetchContent)
FetchContent_Declare(scintilla URL https://www.scintilla.org/scintilla567.zip URL_HASH SHA256=2309181e59e7da4659f890dbc128db7451ccd094748770e7be674b830494587c)
FetchContent_Declare(lexilla URL https://www.scintilla.org/lexilla554.zip URL_HASH SHA256=5e0a21356e4e2f19d6a38ecd1c8172d5a2b7b4542940568b023386080db90fe0)
FetchContent_Declare(pcre2 URL https://github.com/PCRE2Project/pcre2/releases/download/pcre2-10.49/pcre2-10.49.zip URL_HASH SHA256=40fa9173e167648fb872beaaeab667c8778eb37fabeefffa99a7753edb1a9479)
FetchContent_MakeAvailable(scintilla lexilla)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_8 ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_16 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_32 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2GREP OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PCRE2_SUPPORT_JIT OFF CACHE BOOL "" FORCE)
set(PCRE2_STATIC_PIC ON CACHE BOOL "" FORCE)
set(PCRE2_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(pcre2)
file(GLOB SCI_SOURCES CONFIGURE_DEPENDS "${scintilla_SOURCE_DIR}/src/*.cxx")
add_library(scintilla_native STATIC ${SCI_SOURCES}
  "${scintilla_SOURCE_DIR}/win32/ScintillaWin.cxx"
  "${scintilla_SOURCE_DIR}/win32/PlatWin.cxx"
  "${scintilla_SOURCE_DIR}/win32/HanjaDic.cxx"
  "${scintilla_SOURCE_DIR}/win32/ListBox.cxx"
  "${scintilla_SOURCE_DIR}/win32/SurfaceGDI.cxx"
  "${scintilla_SOURCE_DIR}/win32/SurfaceD2D.cxx")
target_include_directories(scintilla_native PUBLIC "${scintilla_SOURCE_DIR}/include" PRIVATE "${scintilla_SOURCE_DIR}/src")
target_compile_definitions(scintilla_native PRIVATE NO_CXX11_REGEX)
target_link_libraries(scintilla_native PUBLIC user32 gdi32 imm32 ole32 oleaut32 advapi32)
file(GLOB LEX_SUPPORT CONFIGURE_DEPENDS "${lexilla_SOURCE_DIR}/lexlib/*.cxx")
add_library(lexers STATIC ${LEX_SUPPORT} editor/lexers.cpp)
foreach(lexer CPP Python JSON HTML CSS Markdown Bash PowerShell Props YAML)
  target_sources(lexers PRIVATE "${lexilla_SOURCE_DIR}/lexers/Lex${lexer}.cxx")
endforeach()
target_include_directories(lexers PRIVATE "${lexilla_SOURCE_DIR}/lexlib" PUBLIC "${lexilla_SOURCE_DIR}/include" "${scintilla_SOURCE_DIR}/include")
add_library(notepad_core STATIC core/core.cpp core/search.cpp)
target_include_directories(notepad_core PUBLIC .)
target_compile_options(notepad_core PRIVATE /W4)
target_compile_definitions(notepad_core PRIVATE PCRE2_STATIC PCRE2_CODE_UNIT_WIDTH=8)
target_link_libraries(notepad_core PUBLIC bcrypt pcre2-8-static)
add_executable(indic-notepad WIN32 app/main.cpp resources/app.rc)
target_compile_options(indic-notepad PRIVATE /W4)
target_link_libraries(indic-notepad PRIVATE notepad_core scintilla_native lexers comctl32 comdlg32 shell32 shlwapi dwmapi uxtheme msimg32)
target_link_options(indic-notepad PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /CETCOMPAT /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest")
enable_testing()
add_executable(notepad-core-tests tests/core_tests.cpp)
target_link_libraries(notepad-core-tests PRIVATE notepad_core)
add_test(NAME document_safety COMMAND notepad-core-tests)
add_executable(notepad-ui-tests WIN32 app/main.cpp resources/app.rc)
target_compile_definitions(notepad-ui-tests PRIVATE NOTEPAD_TEST_BUILD)
target_link_libraries(notepad-ui-tests PRIVATE notepad_core scintilla_native lexers comctl32 comdlg32 shell32 shlwapi dwmapi uxtheme msimg32)
target_link_options(notepad-ui-tests PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest")
install(TARGETS indic-notepad RUNTIME DESTINATION .)
install(FILES "../LICENSE" "../THIRD_PARTY_NOTICES.md" "../docs/HELP.md" DESTINATION .)
install(FILES "${scintilla_SOURCE_DIR}/License.txt" DESTINATION licenses RENAME Scintilla.txt)
install(FILES "${lexilla_SOURCE_DIR}/License.txt" DESTINATION licenses RENAME Lexilla.txt)
install(FILES "${pcre2_SOURCE_DIR}/LICENCE.md" DESTINATION licenses RENAME PCRE2.md)
